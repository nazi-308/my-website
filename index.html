const c = document.getElementById('c'); const ctx = c.getContext('2d'); const save = document.getElementById('save'); let img = null, mode = 'original'; const ids = ['brightness','contrast','saturate','grayscale','sepia','blur']; const v = id => document.getElementById(id).value; function filterStr() { return `brightness(${v('brightness')}%) contrast(${v('contrast')}%) saturate(${v('saturate')}%) grayscale(${v('grayscale')}%) sepia(${v('sepia')}%) blur(${v('blur')}px)`; } function drawPreview() { if (!img) return; const s = Math.min(1, 900 / img.width); c.width = Math.round(img.width * s); c.height = Math.round(img.height * s); ctx.filter = filterStr(); ctx.drawImage(img, 0, 0, c.width, c.height); } function setMode(m, label) { mode = m; save.textContent = 'Download (' + label + ')'; } document.getElementById('file').onchange = e => { const f = e.target.files[0]; if (!f) return; img = new Image(); img.onload = drawPreview; img.src = URL.createObjectURL(f); }; ids.forEach(id => document.getElementById(id).oninput = drawPreview); document.getElementById('x1').onclick = () => setMode('original', 'original'); document.getElementById('x2').onclick = () => setMode('2x', '2x'); document.getElementById('k4').onclick = () => setMode('4k', '4K'); save.onclick = () => { if (!img) return; let s = 1; if (mode === '2x') s = 2; if (mode === '4k') s = 3840 / img.width; const out = document.createElement('canvas'); out.width = Math.round(img.width * s); out.height = Math.round(img.height * s); const o = out.getContext('2d'); o.imageSmoothingEnabled = true; o.imageSmoothingQuality = 'high'; o.filter = filterStr(); o.drawImage(img, 0, 0, out.width, out.height); out.toBlob(b => { const a = document.createElement('a'); a.href = URL.createObjectURL(b); a.download = 'enhanced.png'; a.click(); }); };
